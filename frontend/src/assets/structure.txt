/*
|--------------------------------------------------------------------------
| Imports
|--------------------------------------------------------------------------
*/
import { useState, useEffect, useMemo, useCallback } from "react";
import Something from "@/components/Something";

/*
|--------------------------------------------------------------------------
| Types (optional)
|--------------------------------------------------------------------------
*/
type Props = {
  id: string;
};

/*
|--------------------------------------------------------------------------
| Constants (outside the component)
|--------------------------------------------------------------------------
*/
const DEFAULT_LIMIT = 10;

/*
|--------------------------------------------------------------------------
| Component
|--------------------------------------------------------------------------
*/
export default function MyComponent({ id }: Props) {

  /*
  |--------------------------------------------------------------------------
  | State
  |--------------------------------------------------------------------------
  */
  const [data, setData] = useState(null);
  const [count, setCount] = useState(0);


  /*
  |--------------------------------------------------------------------------
  | Memos
  |--------------------------------------------------------------------------
  */
  const computedValue = useMemo(() => {
    return data ? data.length * 2 : 0;
  }, [data]);


  /*
  |--------------------------------------------------------------------------
  | Callbacks
  |--------------------------------------------------------------------------
  */
  const fetchData = useCallback(async () => {
    const res = await fetch(`/api/items/${id}`);
    const json = await res.json();
    setData(json);
  }, [id]);


  const increment = useCallback(() => {
    setCount((prev) => prev + 1);
  }, []);


  /*
  |--------------------------------------------------------------------------
  | Effects
  |--------------------------------------------------------------------------
  */
  useEffect(() => {
    fetchData();
  }, [fetchData]);


  /*
  |--------------------------------------------------------------------------
  | Render Helpers  (functions used only for UI)
  |--------------------------------------------------------------------------
  */
  const renderHeader = () => (
    <h1 className="text-xl font-bold">Item: {id}</h1>
  );

  const renderList = () => (
    <ul>
      {data?.map((item: any) => (
        <li key={item.id}>{item.name}</li>
      ))}
    </ul>
  );


  /*
  |--------------------------------------------------------------------------
  | JSX RETURN
  |--------------------------------------------------------------------------
  */
  return (
    <div className="space-y-4">
      {renderHeader()}

      <button onClick={increment}>Count: {count}</button>

      <div>Computed Value: {computedValue}</div>

      {renderList()}
    </div>
  );
}

/*
|--------------------------------------------------------------------------
| Utility Functions (not using state)
|--------------------------------------------------------------------------
*/
function formatName(name: string) {
  return name.toUpperCase();
}
